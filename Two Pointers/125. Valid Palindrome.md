class Solution {
public:
    bool isPalindrome(string s) {
        std::transform(s.begin(),s.end(),s.begin(),::tolower);

        string str="";
        for(int i = 0; i < s.size(); i++){
            if(isalnum(s[i])){
                str += s[i];
            }
        }

        int l = 0, r = str.size() - 1;
        while(l < r){
            if(str[l] != str[r]) return false;
            l++;
            r--;
        }
        return true;
    }
};
